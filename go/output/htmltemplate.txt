<script>
let currentPage = 0;
let pages = document.querySelectorAll('.svgContent');

function showPage(page) {
    if (page >= 0 && page < pages.length) {
        const svgContentDisplay = document.getElementById('svgContentDisplay');

        // Remove the old SVG and clear pan-zoom controls if necessary
        if (svgContentDisplay.querySelector('svg')) {
            let oldSvgPanZoomInstance = svgPanZoom(svgContentDisplay.querySelector('svg'));
            if (oldSvgPanZoomInstance) {
                oldSvgPanZoomInstance.destroy();
            }
        }

        // Update the innerHTML with the new SVG content
        svgContentDisplay.innerHTML = pages[page].innerHTML;

        document.querySelectorAll('g.edge.resultEdge').forEach(node => {
          node.addEventListener('click', function() {
              const explanation = node.querySelector('explanation').textContent;
              let formattedExplanation = explanation.replace(/\\n/g, '<br>'); // Replace newlines with HTML line breaks
              formattedExplanation = formattedExplanation.replace(/\\t/g, '&nbsp;&nbsp;&nbsp;&nbsp;'); // Replace tabs with HTML spaces
              document.getElementById('explanationText').innerHTML = formattedExplanation;
          });
        });

        // Reinitialize svg-pan-zoom on the new SVG
        if (svgContentDisplay.querySelector('svg')) {
            svgPanZoom(svgContentDisplay.querySelector('svg'), {
                zoomEnabled: true,
                controlIconsEnabled: true,
                fit: true,
                center: false
            });
        }

        document.getElementById('pageNumber').textContent = page + 1;
        currentPage = page;

        // Update button states
        document.getElementById('prevPage').disabled = currentPage === 0;
        document.getElementById('nextPage').disabled = currentPage === pages.length - 1;
    }
}

function prevPage() {
    if (currentPage > 0) {
        document.getElementById('explanationText').innerHTML = "";
        showPage(currentPage - 1);
    }
}

function nextPage() {
    if (currentPage < pages.length - 1) {
        document.getElementById('explanationText').innerHTML = "";
        showPage(currentPage + 1);
    }
}

function showHelp() {
  document.getElementById('explanationText').innerHTML = document.getElementById('helpContent').innerHTML;
}

document.addEventListener("DOMContentLoaded", function() {
    document.getElementById('totalPages').textContent = pages.length;
    showPage(currentPage); // Show the first page on load

    document.getElementById('prevPage').disabled = currentPage === 0;
    document.getElementById('nextPage').disabled = currentPage === pages.length - 1 || pages.length === 1;

    showHelp() // Show help on load
});
</script>
